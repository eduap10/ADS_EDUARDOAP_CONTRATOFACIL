Campos dinâmicos

As tags permanecem no editor. Na visualização, serão preenchidas automaticamente.

function inserirTag(tag) { var widget = PF('documentoEditor'); if (!widget || !widget.editor) return; var editor = widget.editor; var sel = editor.getSelection(); var i = sel ? sel.index : editor.getLength() - 1; editor.insertText(i, tag, 'user'); editor.setSelection(i + tag.length, 0, 'user'); }
